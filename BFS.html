BFS (广度优先搜索) 算法动态图解

队列 Q: [ 空 ]
变量: 就绪
visited: []
bool visited[MaxVertexNum];
void BFS(ALGraph G, int v) {
    InitQueue(Q);
    visit(v);
    visited[v] = true;
    EnQueue(Q, v);
    while (!IsEmpty(Q)) {
        DeQueue(Q, v);
        ArcNode *p = G.vertices[v].first; 
        while (p != NULL) {
            int w = p->adjvex;
            if (!visited[w]) {
                visit(w);
                visited[w] = true;
                EnQueue(Q, w);
            }
            p = p->next;
        }
    }
}
